cmake_minimum_required(VERSION 3.16)
project(aosl_s31_tasks C)
get_filename_component(repo_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(COMPONENT_PATH "${repo_root}/components/mybot_stack/aosl")
set(COMPONENT_LIB aosl_host_tasks)
set(thread_source "${COMPONENT_PATH}/platform/src/esp32-s3/aosl_hal_thread.c")
set(adapter_source "${COMPONENT_PATH}/s31_port/aosl_task_stack.c")

# Exercise the production wrapper's source-scoped forced include while building
# only the unchanged thread HAL and the project adapter against host mocks.
function(idf_component_register)
    cmake_parse_arguments(component "WHOLE_ARCHIVE" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS;REQUIRES;PRIV_REQUIRES" ${ARGN})
    if(NOT adapter_source IN_LIST component_SRCS OR NOT thread_source IN_LIST component_SRCS)
        message(FATAL_ERROR "The production AOSL component must register both task sources")
    endif()
    add_library(${COMPONENT_LIB} STATIC "${thread_source}" "${adapter_source}")
    target_include_directories(${COMPONENT_LIB} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/mocks"
        "${COMPONENT_PATH}/platform/include" "${COMPONENT_PATH}/s31_port/config")
endfunction()
include("${COMPONENT_PATH}/CMakeLists.txt")
get_source_file_property(adapter_options "${adapter_source}" COMPILE_OPTIONS)
if(adapter_options)
    message(FATAL_ERROR "The project adapter must not redirect its own FreeRTOS calls")
endif()
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_compile_definitions(${COMPONENT_LIB} PRIVATE _POSIX_C_SOURCE=200809L)
add_executable(test_tasks test_tasks.c)
target_link_libraries(test_tasks PRIVATE ${COMPONENT_LIB} pthread)
target_include_directories(test_tasks PRIVATE "${COMPONENT_PATH}/s31_port")
target_compile_options(test_tasks PRIVATE -Wall -Wextra -Werror)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${COMPONENT_LIB} PRIVATE -fsanitize=undefined)
    target_compile_options(test_tasks PRIVATE -fsanitize=undefined)
    target_link_options(test_tasks PRIVATE -fsanitize=undefined)
endif()
set_target_properties(${COMPONENT_LIB} test_tasks PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
